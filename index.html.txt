<script>
// Variable global para evitar duplicar/reinicializar el mapa en múltiples clics
let mapa = null;

function calcularTotal() {
  let precioBase = parseFloat(document.getElementById("tamaño").value);
  let cantidad = parseInt(document.getElementById("cantidad").value) || 1;
  let complemento = document.getElementById("complemento").checked ? 399 : 0;

  // Aplicar 5% de descuento a las playeras si la cantidad es >= 20
  if (cantidad >= 20) {
    precioBase = precioBase * 0.95;
  }

  // Desglose correcto:
  // Subtotal = (Precio Unitario Descontado * Cantidad) + Complemento
  let subtotalProductos = (precioBase * cantidad) + complemento;
  let envio = 150;
  let baseImponible = subtotalProductos + envio;
  
  let iva = baseImponible * 0.16;
  let total = baseImponible + iva;

  document.getElementById("total").innerHTML = "<strong>Total: $" + total.toFixed(2) + " MXN</strong>";
  return { precioBase, cantidad, complemento, envio, subtotalProductos, iva, total };
}

function mostrarFormulario() {
  document.getElementById("formulario").style.display = "block";
  calcularTotal();
}

function emitirTicket() {
  let datos = calcularTotal();
  let nombre = document.getElementById("nombreCliente").value;
  let direccion = document.getElementById("direccionEnvio").value;

  let detalle = `
    Cliente: ${nombre}<br>
    Dirección de envío: ${direccion}<br><br>
    Producto: Playera Tigres UANL 2024–2025<br>
    Precio unitario: $${datos.precioBase.toFixed(2)} MXN<br>
    Cantidad: ${datos.cantidad}<br>
    Complemento: ${datos.complemento > 0 ? "Gorra Tigres $399 MXN" : "Ninguno"}<br>
    Envío fijo: $${datos.envio.toFixed(2)} MXN<br>
    IVA (16%): $${datos.iva.toFixed(2)} MXN<br>
    <strong>Total: $${datos.total.toFixed(2)} MXN</strong>
  `;
  document.getElementById("detalle").innerHTML = detalle;
  document.getElementById("ticket").style.display = "block";

  // Reiniciar el contenedor del mapa si ya existía para evitar el error "Map container is already initialized"
  if (mapa !== null) {
    mapa.remove();
  }

  // Inicializar mapa
  mapa = L.map('mapa').setView([25.813, -100.597], 10); // García, NL
  L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
    attribution: '© OpenStreetMap contributors'
  }).addTo(mapa);

  // Geocodificación con Nominatim
  fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(direccion)}`)
    .then(response => response.json())
    .then(data => {
      if (data && data.length > 0) {
        let lat = parseFloat(data[0].lat);
        let lon = parseFloat(data[0].lon);

        L.Routing.control({
          waypoints: [
            L.latLng(25.813, -100.597), // García, NL
            L.latLng(lat, lon)
          ],
          routeWhileDragging: true
        }).addTo(mapa);

        L.marker([25.813, -100.597]).addTo(mapa).bindPopup("Producto en García, NL").openPopup();
        L.marker([lat, lon]).addTo(mapa).bindPopup("Dirección de entrega").openPopup();
      } else {
        alert("No se pudo localizar la dirección ingresada.");
      }
    });
}

// Ejecutar al cargar la página para inicializar el total
window.onload = calcularTotal;
</script>
